extends GutTest

const MAP_SCENE: PackedScene = preload("res://scenes/maps/museum_v2.tscn")
const FINISHES: MuseumFinishes = preload("res://data/visual/museum_finishes.tres")
const OUT: String = "user://p189_%s_%s_%02d.png"
const SPOTS: Dictionary = {"vestibulo": [24.0, 31.0], "retratos": [8.0, 21.0], "servicio": [33.0, 22.0]}
const SKYLIGHTS: Array[String] = ["VES", "GW", "GSW", "ANT", "PAT", "ORI"]
const WALK: int = 24
const STEP: float = 0.07
const BLUE: Color = Color("2b3354")
const WINE: Color = Color("4a2832")
const GREEN: Color = Color("24403a")


func _hide_ui() -> void:
	for node: Node in get_tree().root.find_children("*", "CanvasLayer", true, false):
		(node as CanvasLayer).visible = false


func _grab(key: String, style: String, i: int) -> void:
	await get_tree().process_frame
	await RenderingServer.frame_post_draw
	get_viewport().get_texture().get_image().save_png(OUT % [style, key, i])


func _materials(map: Node) -> Array[ShaderMaterial]:
	var found: Dictionary = {}
	for node: Node in map.find_children("*", "GeometryInstance3D", true, false):
		var mats: Array = [(node as GeometryInstance3D).material_override]
		if node is MeshInstance3D:
			var mi: MeshInstance3D = node as MeshInstance3D
			if mi.mesh != null:
				for i: int in range(mi.mesh.get_surface_count()):
					mats.append(mi.get_surface_override_material(i))
					mats.append(mi.mesh.surface_get_material(i))
				if mi.mesh is PrimitiveMesh:
					mats.append((mi.mesh as PrimitiveMesh).material)
		for m: Variant in mats:
			var mat: Material = m as Material
			while mat != null:
				if mat is ShaderMaterial:
					found[mat] = true
				mat = mat.next_pass
	var out: Array[ShaderMaterial] = []
	out.assign(found.keys())
	return out


func _palette() -> void:
	var map_colors: Dictionary = {
		"VES": BLUE, "ROT": WINE, "ANTE": WINE, "ANT": GREEN, "GW": WINE, "GSW": BLUE,
		"ORI": GREEN, "BIB": GREEN, "PAT": BLUE, "TAP": WINE, "OFF": BLUE, "JOY": WINE,
	}
	var f: MuseumFinishes = FINISHES
	f.paints = map_colors
	f.dado = Color("221a18")
	f.service_wall = Color("3c3f46")
	f.vault_wall = Color("4a505c")


func _apply(map: Node, style: String) -> void:
	# Luz suave y natural para las tres (Q35-Q38)
	for mat: ShaderMaterial in _materials(map):
		if mat.shader != null and "band_softness" in str(mat.shader.get_shader_uniform_list()):
			mat.set_shader_parameter("band_softness", 0.15)
		if style != "B":
			var name: String = mat.shader.resource_path if mat.shader else ""
			if name.ends_with("museum_wall.gdshader"):
				mat.set_shader_parameter("pattern_strength", 0.0)
			elif name.ends_with("museum_floor.gdshader"):
				mat.set_shader_parameter("pattern_strength", 0.25)
		else:
			mat.set_shader_parameter("pattern_strength", 0.5)
	for node: Node in map.find_children("*", "Light3D", true, false):
		(node as Light3D).shadow_blur = 3.0
	for node: Node in map.find_children("*", "DirectionalLight3D", true, false):
		var moon: DirectionalLight3D = node as DirectionalLight3D
		moon.light_color = Color(0.55, 0.65, 1.0)
	var data: LevelData = map.get_meta(&"level_data") as LevelData
	for room_id: String in SKYLIGHTS:
		var rect: Rect2 = data.room_rect(data.room_by_id(room_id))
		var c: Vector2 = rect.get_center() + data.offset
		var sky: SpotLight3D = SpotLight3D.new()
		sky.name = "Skylight_%s" % room_id
		sky.light_color = Color(0.5, 0.62, 1.0)
		sky.light_energy = 2.2
		sky.spot_range = 12.0
		sky.spot_angle = 16.0
		sky.spot_angle_attenuation = 0.6
		sky.shadow_enabled = true
		sky.shadow_blur = 3.0
		map.add_child(sky)
		sky.global_position = Vector3(c.x - 1.5, 9.0, c.y + 1.0)
		sky.rotation_degrees = Vector3(-80.0, 20.0, 0.0)
	var fog: Node = map.find_children("FogOfWar", "", true, false)[0]
	var f: ShaderMaterial = fog.get("_material") as ShaderMaterial
	f.set_shader_parameter("line_min_height", 2.0)
	match style:
		"A":
			f.set_shader_parameter("ink_floor_dark", Color(0.01, 0.015, 0.035))
			f.set_shader_parameter("ink_floor_light", Color(0.03, 0.045, 0.09))
			f.set_shader_parameter("ink_raised_dark", Color(0.035, 0.05, 0.1))
			f.set_shader_parameter("ink_raised_light", Color(0.07, 0.1, 0.18))
			f.set_shader_parameter("ink_line", Color(0.45, 0.58, 0.85))
			f.set_shader_parameter("line_strength", 0.35)
		"B":
			f.set_shader_parameter("line_strength", 0.2)
		"C":
			f.set_shader_parameter("ink_floor_dark", Color(0.005, 0.01, 0.015))
			f.set_shader_parameter("ink_floor_light", Color(0.015, 0.03, 0.04))
			f.set_shader_parameter("ink_raised_dark", Color(0.02, 0.04, 0.05))
			f.set_shader_parameter("ink_raised_light", Color(0.04, 0.08, 0.1))
			f.set_shader_parameter("ink_line", Color(0.18, 0.95, 1.0))
			f.set_shader_parameter("line_strength", 0.45)


func test_direcciones() -> void:
	var style: String = OS.get_environment("P189_STYLE")
	if style != "B":
		_palette()
	var map: HeistMap3D = MAP_SCENE.instantiate() as HeistMap3D
	get_tree().root.add_child(map)
	await wait_physics_frames(5)
	for guard: GuardController3D in map.guards:
		guard.set_physics_process(false)
	map.thief.set_physics_process(false)
	_apply(map, style)
	var data: LevelData = map.get_meta(&"level_data") as LevelData
	var offset: Vector3 = Vector3(data.offset.x, 0.0, data.offset.y)
	for key: String in SPOTS:
		var spot: Array = SPOTS[key]
		var start: Vector3 = offset + Vector3(spot[0], 0.2, spot[1])
		map.thief.global_position = start
		map.camera.global_position = start + HeistMap3D.CAMERA_OFFSET
		_hide_ui()
		await wait_seconds(2.5)
		for i: int in range(WALK):
			map.thief.global_position = start + Vector3(STEP, 0.0, -STEP) * float(i)
			await _grab(key, style, i)
	map.queue_free()
	await wait_physics_frames(2)
	assert_true(true)
